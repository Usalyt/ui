## Usage

```tsx
// components/ui/blocks/heuristic-finding-card.tsx
import { HeuristicFindingCard } from "@usalyt/ui";

export function FindingRecord() {
  return (
    <HeuristicFindingCard
      finding={{
        id: "finding-h1-01",
        heuristicId: "H1",
        heuristicTitle: "Visibility of System Status",
        severity: "CRITICAL",
        rating: 4,
        selector: "button#complete-purchase.btn-primary",
        description: "The button shows no loading state while the transaction is processed.",
      }}
      onDelete={(id) => console.log("delete", id)}
    />
  );
}
```
