## Usage

```tsx
// components/ui/collaboration-chat-panel/collaboration-chat-panel.tsx
import { CollaborationChatPanel } from "tu-ui";

export function ProjectChat() {
  return (
    <CollaborationChatPanel
      projectName="Checkout Heuristic Audit #4"
      onlineCount={3}
      messages={[
        {
          id: "msg-1",
          senderName: "Marcos Lead",
          content: "Please review the checkout button contrast.",
          timestamp: "11:20 AM",
          attachedFindingId: "H1-03",
        },
      ]}
      onSendMessage={(content) => console.log(content)}
    />
  );
}
```
