## Usage

```tsx
// components/ui/blocks/app-shell-layout.tsx
import { AppShellLayout } from "@usalyt/ui";

export function MyAppShell() {
  return (
    <AppShellLayout
      brandName="UX Check Platform"
      navItems={[{ title: "Overview", href: "#", isActive: true }]}
      breadcrumbs={[{ label: "Projects", href: "#" }, { label: "Checkout Flow Audit" }]}
      user={{ name: "Alex Doe", email: "alex@uxcheck.io" }}
    >
      <h2 className="text-xl font-bold">Project Dashboard</h2>
    </AppShellLayout>
  );
}
```
