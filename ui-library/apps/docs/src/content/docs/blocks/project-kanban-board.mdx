## Usage

```tsx
// components/ui/project-kanban-board/project-kanban-board.tsx
import { ProjectKanbanBoard, type KanbanTask } from "tu-ui";

const tasks: KanbanTask[] = [
  {
    id: "task-01",
    title: "Adjust contrast ratio on secondary actions",
    status: "TODO",
    heuristicTag: "H1",
    severity: "CRITICAL",
    assigneeName: "Lucas",
  },
];

export function RemediationBoard() {
  return (
    <ProjectKanbanBoard
      tasks={tasks}
      onTaskMove={(taskId, targetStatus) => console.log(taskId, targetStatus)}
      onTaskClick={(task) => console.log("open", task.id)}
      onAddTask={(status) => console.log("add to", status)}
    />
  );
}
```

## Props

<PropsTable props={[
  { name: "tasks", type: "KanbanTask[]", defaultValue: "[]", description: "Board tasks ({ id, title, status, heuristicTag?, severity?, assigneeName?, assigneeAvatar? }). Empty columns render an explicit empty state." },
  { name: "onTaskMove", type: "(taskId: string, targetStatus: KanbanColumnId) => void", defaultValue: "—", description: "Renders a per-task move menu; selecting a column calls this handler without requiring drag-and-drop." },
  { name: "onTaskClick", type: "(task: KanbanTask) => void", defaultValue: "—", description: "Renders the task title as a button that activates with click, Enter, or Space." },
  { name: "onAddTask", type: "(status: KanbanColumnId) => void", defaultValue: "—", description: "Renders the per-column add button and handles clicks." },
]} />
