Profile settings Block composed from Card, Form, Input, and Button. Submission results may be asynchronous: the footer action exposes a pending state and rejected promises render a `role="alert"` message.

## Usage

```tsx
// components/ui/form-with-validation/form-with-validation.tsx
import { FormWithValidation, type ProfileFormValues } from "tu-ui";

export function ProfileSettings() {
  return (
    <FormWithValidation
      defaultValues={{ fullName: "", email: "", organization: "" }}
      submitLabel="Update Profile"
      onSubmit={async (values: ProfileFormValues) => {
        await saveProfile(values);
      }}
    />
  );
}
```
