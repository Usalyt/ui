File selection primitive rendered as a clickable dropzone around a native file input. The hidden input keeps native behavior while the wrapper exposes a visible focus ring when keyboard focus enters it.

## Usage

```tsx
// components/ui/file-input/file-input.tsx
import { FileInput } from "tu-ui";

export function EvidenceUpload() {
  return (
    <FileInput
      label="Upload evaluation evidence"
      helperText="PNG, JPG or PDF up to 10MB"
      accept="image/*,.pdf"
      onChange={(event) => console.log(event.target.files)}
    />
  );
}
```
