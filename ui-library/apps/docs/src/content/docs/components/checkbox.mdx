## Usage

```tsx
// components/ui/checkbox.tsx
import { Checkbox, Label } from "@usalyt/ui";

export function CheckboxExample() {
  return (
    <div className="flex items-center space-x-2">
      <Checkbox id="terms" />
      <Label htmlFor="terms">Accept terms and conditions</Label>
    </div>
  );
}
```
