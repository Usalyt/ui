## Usage

```tsx
// components/ui/avatar.tsx
import { Avatar, AvatarImage, AvatarFallback } from "tu-ui";

export function AvatarExample() {
  return (
    <Avatar>
      <AvatarImage src="https://github.com/shadcn.png" alt="@shadcn" />
      <AvatarFallback>CN</AvatarFallback>
    </Avatar>
  );
}
```
