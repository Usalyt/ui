Collapsible tree built from `TreeView`, `TreeItem`, `TreeItemTrigger`, and `TreeItemContent`. Items render `role="treeitem"` with `aria-expanded`, and nesting is plain composition — put a `TreeItem` inside another item's `TreeItemContent`.

## Usage

```tsx
// components/ui/tree-view.tsx
import { TreeView, TreeItem, TreeItemTrigger, TreeItemContent } from "@usalyt/ui";

export function FileTree() {
  return (
    <TreeView>
      <TreeItem defaultExpanded>
        <TreeItemTrigger>src</TreeItemTrigger>
        <TreeItemContent>
          <TreeItem>
            <TreeItemTrigger>components</TreeItemTrigger>
            <TreeItemContent>
              <TreeItem>
                <span>button.tsx</span>
              </TreeItem>
            </TreeItemContent>
          </TreeItem>
        </TreeItemContent>
      </TreeItem>
    </TreeView>
  );
}
```
