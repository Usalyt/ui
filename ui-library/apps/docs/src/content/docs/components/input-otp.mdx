One-time-code input built on the `input-otp` library. `InputOTPSlot` elements read their character and caret state from position `index`, so group and decorate them freely.

## Usage

```tsx
// components/ui/input-otp.tsx
import { InputOTP, InputOTPGroup, InputOTPSlot, InputOTPSeparator } from "tu-ui";

export function VerificationCode() {
  return (
    <InputOTP maxLength={6}>
      <InputOTPGroup>
        <InputOTPSlot index={0} />
        <InputOTPSlot index={1} />
        <InputOTPSlot index={2} />
      </InputOTPGroup>
      <InputOTPSeparator />
      <InputOTPGroup>
        <InputOTPSlot index={3} />
        <InputOTPSlot index={4} />
        <InputOTPSlot index={5} />
      </InputOTPGroup>
    </InputOTP>
  );
}
```
