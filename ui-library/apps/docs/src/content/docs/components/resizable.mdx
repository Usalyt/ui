Draggable panel layout built on `react-resizable-panels`. Compose `ResizablePanelGroup`, `ResizablePanel`, and `ResizableHandle`; handles are focusable and keyboard-accessible.

## Usage

```tsx
// components/ui/resizable.tsx
import { ResizablePanelGroup, ResizablePanel, ResizableHandle } from "@usalyt/ui";

export function WorkspacePanes() {
  return (
    <ResizablePanelGroup direction="horizontal">
      <ResizablePanel defaultSize={25} minSize={20}>Navigation</ResizablePanel>
      <ResizableHandle withHandle />
      <ResizablePanel defaultSize={75}>Preview</ResizablePanel>
    </ResizablePanelGroup>
  );
}
```
