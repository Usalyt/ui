## Usage

```tsx
// components/ui/kbd.tsx
import { Kbd } from "@usalyt/ui";

export function ShortcutHint() {
  return (
    <p className="text-sm text-muted-foreground">
      Open search with <Kbd size="md">⌘K</Kbd>
    </p>
  );
}
```
