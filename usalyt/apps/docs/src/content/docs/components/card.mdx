## Usage

```tsx
// components/ui/card.tsx
import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter, Button } from "@usalyt/ui";

export function CardExample() {
  return (
    <Card className="w-80">
      <CardHeader>
        <CardTitle>Card Title</CardTitle>
        <CardDescription>Card Description</CardDescription>
      </CardHeader>
      <CardContent>
        <p>Card Content</p>
      </CardContent>
      <CardFooter>
        <Button className="w-full">Action</Button>
      </CardFooter>
    </Card>
  );
}
```
