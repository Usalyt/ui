## Usage

```tsx
// components/ui/blocks/team-members-table.tsx
import { TeamMembersTable, type TeamMember } from "@usalyt/ui";

const members: TeamMember[] = [
  { id: "usr-01", name: "Sara Gómez", email: "sara@example.com", role: "owner", joinedAt: "12 May 2026" },
  { id: "usr-02", name: "Carlos Rivera", email: "carlos@example.com", role: "evaluator", joinedAt: "18 May 2026" },
];

export function ProjectMembers() {
  return (
    <TeamMembersTable
      members={members}
      onRoleChange={(memberId, newRole) => console.log(memberId, newRole)}
      onRemoveMember={(memberId) => console.log("remove", memberId)}
      onInviteMember={() => console.log("invite")}
    />
  );
}
```
