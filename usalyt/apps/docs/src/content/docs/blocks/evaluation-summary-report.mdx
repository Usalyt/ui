## Usage

```tsx
// components/ui/blocks/evaluation-summary-report.tsx
import { EvaluationSummaryReport } from "@usalyt/ui";

export function AuditSummary() {
  return (
    <EvaluationSummaryReport
      title="Checkout Flow Usability Audit"
      projectName="E-Commerce Web Portal"
      overallScore={78}
      criticalCount={4}
      mediumCount={6}
      goodCount={12}
      heuristics={[
        { id: "H1", name: "Visibility of system status", score: 75, findingCount: 3 },
      ]}
      onExportPdf={() => console.log("export")}
    />
  );
}
```
