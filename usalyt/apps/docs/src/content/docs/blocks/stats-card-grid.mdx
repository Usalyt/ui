Responsive metric grid with titled cards, mono values, trend badges, and supporting copy. When no stats are supplied the Block renders an explicit empty state instead of an empty grid.

## Usage

```tsx
// components/ui/stats-card-grid/stats-card-grid.tsx
import { StatsCardGrid, type StatItem } from "@usalyt/ui";

const stats: StatItem[] = [
  {
    id: "stat-01",
    title: "Findings logged",
    value: 24,
    change: "+12%",
    trend: "up",
  },
];

export function DashboardStats() {
  return <StatsCardGrid stats={stats} columns={4} />;
}
```
